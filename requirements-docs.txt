# Build-time dependencies for the MkDocs documentation site.
# Pin to known-good versions; bump quarterly or when a security
# advisory lands.
#
# Install with: python -m pip install -r requirements-docs.txt
# Build with:   mkdocs build --strict
# Serve with:   mkdocs serve   (live-reload at http://localhost:8000)

mkdocs==1.6.1
mkdocs-material==9.7.7
mkdocs-include-markdown-plugin==7.3.0
# pymdown-extensions 10.11.2 + pygments 2.19 raised:
#   AttributeError: 'NoneType' object has no attribute 'replace'
# from pygments.HtmlFormatter when an unspecified `filename=` was
# forwarded as None. Bumping to 10.12+ where pymdownx coerces the
# value before handing it to pygments fixes the build under
# whatever pygments the runner pulls in.
pymdown-extensions==11.0.2
# Pin pygments too so a future-runner upgrade doesn't reopen the
# same regression: 2.18 is the last version before the strictness
# change that interacted with old pymdownx; 2.19 is the version
# the failing runner used. Either is fine with pymdownx 10.12+.
pygments==2.21.0
